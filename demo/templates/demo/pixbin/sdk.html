get_pixbin_client()

Get a configured Pixbin client instance from Django settings

{# djlint:off #} 
from django_pixbin import get_pixbin_client

# Uses PIXBIN_API_TOKEN and PIXBIN_BASE_URL from settings
client = get_pixbin_client()

# Upload a path or any file-like object (Django UploadedFile, BytesIO, ...)
image_id = client.upload_file(request.FILES["image"])
image_id = client.upload_file("photo.jpg", caption="Product photo", private=True)

# upload_file waits for processing by default; pass max_wait=0 to return
# right after the upload and let pixbin.js poll in the browser
image_id = client.upload_file(io.BytesIO(image_bytes), max_wait=0)
{# djlint:on #}

Download Functions

Download original or transformed image bytes

{# djlint:off #} 
# Download original image
original_bytes = client.download_original(image_id)

# Download transformed image
transformed_bytes = client.download_transformed(
    image_id,
    'resize:300x300:fit,quality:85'
)

# Save to file or return as response (sniff the format, it may be webp or png)
with open('output.jpg', 'wb') as f:
    f.write(transformed_bytes)

Helper Functions

Convenience functions for common transformation patterns. The parameter strings below are generated by calling these helpers in the view.

{# djlint:off #} 
from pixbin import thumbnail, crop_square, optimize_web

# Generate transformation parameters
params = thumbnail(300, 300, quality=85)
# Returns: 'resize:300x300:fit,quality:85'

params = crop_square(200, 'center')
# Returns: 'crop:center,resize:200x200:exact'

params = optimize_web(max_width=1920, quality=85)
# Returns: 'resize:1920x1920:fit,quality:85,format:webp'

# Use with transform URL
url = client.transform_url(image_id, params)

Error Handling

Exception classes for handling different error scenarios

{# djlint:off #} 
from pixbin import (
    PixbinError,       # Base exception
    PixbinAuthError,   # Invalid/missing API token
    PixbinQuotaError,  # Storage/bandwidth quota exceeded
    PixbinUploadError, # Upload failed
)

try:
    image_id = client.upload_file(file)
except PixbinAuthError:
    # Handle authentication failure
    messages.error(request, "Invalid API token")
except PixbinQuotaError:
    # Handle quota exceeded
    messages.error(request, "Storage quota exceeded")
except PixbinUploadError as e:
    # Handle upload failure
    messages.error(request, f"Upload failed: {e}")
except PixbinError as e:
    # Handle any other Pixbin error
    messages.error(request, f"Error: {e}")
{# djlint:on #}
