document.querySelectorAll('img').forEach(img => {
    img.addEventListener('pixbin:processing', () => {
        console.log('Image is processing...');
    });
    img.addEventListener('pixbin:ready', () => {
        console.log('Image is ready!');
    });
    img.addEventListener('pixbin:timeout', () => {
        console.log('Image processing timed out');
    });
